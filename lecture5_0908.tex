\HandoutMarkSection{1}{The Discrete-Time Fourier Transform (DTFT)}

\begin{defbox}[title={Def}]
The DTFT $X(e^{j\omega})$ of a sequence $x[n]$ is given by
\begin{equation}
X(e^{j\omega}) = \sum_{n=-\infty}^{\infty} x[n]\,e^{-j\omega n}.\footnotemark{}\MitraTagRef{3.10}
\end{equation}
$\Rightarrow$ a complex-valued function of a real-valued variable $\omega$.
\begin{equation}
X(e^{j\omega}) = X_{\mathrm{re}}(e^{j\omega}) + j X_{\mathrm{im}}(e^{j\omega}).
\end{equation}
Both $X_{\mathrm{re}}(e^{j\omega})$ and $X_{\mathrm{im}}(e^{j\omega})$ are real-valued functions of $\omega$.
\end{defbox}
\footnotetext{Mitra: Equation~(3.10) usually is referred to as the \textbf{analysis equation}, because it analyzes how much of each complex exponential signal is present in the original signal.}

\begin{equation}
X(e^{j\omega}) = \underbrace{|X(e^{j\omega})|}_{\HandoutUnderbraceLabel{\text{magnitude}}}\,e^{j\HandoutHighlightYellow{$\theta(\omega)$}}, \qquad \theta(\omega) = \angle X(e^{j\omega})\ \text{(\textbf{phase})}.
\end{equation}

\begin{notebox}
Phase is not unique:
\begin{equation*}
  e^{j\theta(\omega)} = e^{j(\theta(\omega)+2\pi k)}, \quad k \in \Z.
\end{equation*}
\noindent Sometimes we force the \textbf{principal value},
$-\pi \le \theta(\omega) < +\pi$.
\end{notebox}

\begin{examplebox}[title={DTFT Examples}, handout mark=2]
\begin{itemize}
  \item $x[n] = \delta[n]$:
  \begin{align*}
    X(e^{j\omega}) &= \sum_{n=-\infty}^{\infty} \delta[n]\,e^{-j\omega n} = \delta[0]\,e^{-j\omega \cdot 0} = 1.
  \end{align*}
  \item $x[n] = \alpha^n u[n]$, $|\alpha|<1$:
  \begin{align*}
    X(e^{j\omega}) &= \sum_{n=-\infty}^{\infty} \alpha^n u[n]\,e^{-j\omega n}
      = \sum_{n=0}^{\infty} \alpha^n e^{-j\omega n} = \sum_{n=0}^{\infty} \left(\alpha e^{-j\omega}\right)^n \\
    &= \frac{1}{1-\alpha e^{-j\omega}} \quad \text{if } |\alpha e^{-j\omega}| < 1 \quad \left(|\alpha e^{-j\omega}| = |\alpha|\,\underbrace{|e^{-j\omega}|}_{\HandoutUnderbraceLabel{1}}\right).
  \end{align*}
\end{itemize}
\end{examplebox}

\begin{examplebox}[title={Example (Rectangular Window Sequence)}]
\begin{align*}
  x[n] &= \begin{cases} 1, & 0\le n \le N-1 \\ 0, & \text{else} \end{cases} \\
  &= u[n] - u[n-N].
\end{align*}%
\HandoutMarkBeforeAlign{3}%
\begin{align*}
  X(e^{j\omega}) &= \sum_{n=0}^{N-1} e^{-j\omega n} \\
  &= \frac{1-\left(e^{-j\omega}\right)^N}{1-e^{-j\omega}} \\
  &= \frac{e^{-j\omega\frac{N}{2}}\left(e^{+j\omega\frac{N}{2}}-e^{-j\omega\frac{N}{2}}\right)}{e^{-j\frac{\omega}{2}}\left(e^{+j\frac{\omega}{2}}-e^{-j\frac{\omega}{2}}\right)} \\
  &= \HandoutHighlightYellow{$\frac{\sin\!\left(\omega\frac{N}{2}\right)}{\sin\!\left(\frac{\omega}{2}\right)}$}\,e^{-j\omega\frac{N-1}{2}}.
\end{align*}
\end{examplebox}

\HandoutMarkSection{2}{Dirichlet Kernel}

\includegraphics[width=\linewidth,height=2.6cm,keepaspectratio]{Peter_Gustav_Lejeune_Dirichlet.jpg}\\[-0.1em]
{\scriptsize\itshape P.~G.~Lejeune Dirichlet}
\vspace{-0.35\baselineskip}
\end{wrapfigure}
In signal processing, the Dirichlet kernel is often called the periodic sinc function:
\[
\frac{\sin\!\left(\omega\frac{N}{2}\right)}{\sin\!\left(\frac{\omega}{2}\right)}.
\]
For small $\omega$, $\sin\!\left(\frac{\omega}{2}\right)\approx \frac{\omega}{2}$, so
\begin{equation}
\frac{\sin\!\left(\omega\frac{N}{2}\right)}{\sin\!\left(\frac{\omega}{2}\right)}
  \approx \frac{\sin\!\left(\omega\frac{N}{2}\right)}{\frac{\omega}{2}}
  = N\,\mathrm{sinc}\!\left(\omega\frac{N}{2}\right),
  \qquad \mathrm{sinc}(x) \triangleq \frac{\sin x}{x}.
\end{equation}
Zeros occur when $\omega\dfrac{N}{2} = k\pi$, $k\ne 0$, i.e., $\omega = \dfrac{2\pi k}{N}$.

\begin{center}
\adjustbox{max width=\linewidth,max height=0.24\textheight,center}{%
  \includegraphics{dirichlet_zeros.pdf}%
}\\[-0.35em]
{\scriptsize ``more zeros'' for larger $N$, faster oscillations.}
\end{center}

\noindent\HandoutMarkLeft{4}\textbf{But:} also periodic (``periodic sinc''):
\begin{align*}
\left|\frac{\sin\!\left(\omega\frac{N}{2}\right)}{\sin\!\left(\frac{\omega}{2}\right)}\right| \ \overset{?}{=}\ \left|\frac{\sin\!\left((\omega+2\pi)\frac{N}{2}\right)}{\sin\!\left(\frac{\omega+2\pi}{2}\right)}\right|
= \left|\frac{\sin\left(\omega\frac{N}{2}+\pi N\right)}{\sin\left(\frac{\omega}{2}+\pi\right)}\right|.
\end{align*}

\begin{center}
\adjustbox{max width=\linewidth,max height=0.22\textheight,center}{%
  \includegraphics{dirichlet_periodic.pdf}%
}\\[-0.35em]
{\scriptsize $|D(\omega)| = |D(\omega + 2\pi)|$}
\end{center}

\textbf{Dealing with $0/0$ when $\omega=0$:}
\begin{equation}
X(e^{j\omega})\Big|_{\omega=0} = \sum_{n=0}^{N-1} e^{-j\omega n}\Big|_{\omega=0} = N.
\end{equation}
Limits: $\displaystyle\lim_{\omega\to 0} \frac{f(\omega)}{g(\omega)}$. For small $\omega$, $\dfrac{\omega\frac{N}{2}}{\frac{\omega}{2}} = N$.

\HandoutMarkSection{5}{DTFT Properties\HandoutTitleFnmark}
\footnotetext{cf. Mitra Sec.~3.2.2, Basic Properties.}

\subsection{Periodicity}

Unlike the continuous-time Fourier transform, the DTFT is a periodic function of $\omega$ with period $2\pi$.\footnotemark{}
\footnotetext{cf. Mitra Sec.~3.2.1.}

\begin{factbox}[title={Property}]
The DTFT is $2\pi$-periodic: $X(e^{j(\omega+2\pi k)}) = X(e^{j\omega})$ for all $k\in\Z$.
\end{factbox}

\paragraph{Proof.}
\begin{align*}
X\!\left(e^{j(\omega_0+2\pi k)}\right) &= \sum_{n=-\infty}^{\infty} x[n]\,e^{-j(\omega_0+2\pi k)n} \\
&= \sum_{n=-\infty}^{\infty} x[n]\,e^{-j\omega_0 n}\,\underbrace{e^{-j2\pi k n}}_{\HandoutUnderbraceLabel{e^{-j2\pi kn}=1}} \\
&= \sum_{n=-\infty}^{\infty} x[n]\,e^{-j\omega_0 n} = X(e^{j\omega_0}), \quad \text{for all values of $k$.} \quad \blacksquare
\end{align*}

\begin{notebox}[title={Remark\hfill\textnormal{\small\textit{(addendum to orig lecture note)}}}]
Why is $e^{-j2\pi kn}=1$? Both $k$ and $n$ are integers, so $kn\in\Z$ and the angle $-2\pi kn$ is an integer multiple of $2\pi$.
Each full turn on the unit circle returns to~$1$:
\begin{equation*}
  e^{-j2\pi kn}=\cos(2\pi kn)-j\sin(2\pi kn)=1.
\end{equation*}
\end{notebox}

\subsection{Inverse transform}

\begin{defbox}[title={Inverse DTFT}]
\begin{equation}
x[n] = \frac{1}{2\pi}\int_{-\pi}^{\pi} X(e^{j\omega})\,e^{+j\omega n}\,d\omega.\footnotemark{}\MitraTag{3.14}
\end{equation}
\end{defbox}
\footnotetext{Mitra:
\begin{itemize}[nosep, leftmargin=1.2em]
  \item It should be noted that even though the integration in Eq.~(3.14) can be carried out over any interval of duration~$2\pi$, it is a common practice to choose the interval $[-\pi,\pi]$.
  \item The inverse discrete-time Fourier transform given by Eq.~(3.14) can be interpreted as a linear combination of infinitesimally small complex exponential signals of the form $\frac{1}{2\pi}\,e^{j\omega n}\,d\omega$, weighted by the complex constant $X(e^{j\omega})$ over the normalized angular frequency range from $-\pi$ to $\pi$.
  \item Eq.~(3.14) is referred to as the \textbf{synthesis equation}, because it synthesizes an arbitrary signal from its complex exponential components.
\end{itemize}}

\noindent Equations~(3.10) and~(3.14) form a DTFT pair. Write
\begin{equation*}
\HandoutHighlightMath{$\mathcal{F}\{x[n]\}$}=X(e^{j\omega}), \qquad
\HandoutHighlightMath{$\mathcal{F}^{-1}\{X(e^{j\omega})\}$}=x[n],
\end{equation*}
and denote the DTFT pair by
\begin{equation}
x[n] \mathrel{\mathop{\longleftrightarrow}^{\mathcal{F}}} X(e^{j\omega}).\MitraTag{3.15}
\end{equation}

\paragraph{Proof.}
\begin{align*}
x[n] &= \frac{1}{2\pi}\int_{-\pi}^{\pi}\left(\sum_{\ell=-\infty}^{\infty} x[\ell]\,e^{-j\omega \ell}\right)e^{j\omega n}\,d\omega \\
&= \sum_{\ell=-\infty}^{\infty} x[\ell]\int_{-\pi}^{\pi}\frac{1}{2\pi}\,e^{j\omega(n-\ell)}\,d\omega \\
&= \sum_{\ell=-\infty}^{\infty} x[\ell]\,\frac{\sin\big(\pi(n-\ell)\big)}{\pi(n-\ell)}
= \sum_{\ell=-\infty}^{\infty} x[\ell]\,\delta[n-\ell] \\
&= x[n]. \quad \blacksquare
\end{align*}

\HandoutMarkSubsection{6}{Convergence\HandoutTitleFnmark}
\footnotetext{cf. Mitra Sec.~3.2.4.}
\label{sec:dtft-convergence}

\begin{notebox}
Exchanging the order of integration and summation (above) assumed convergence.
\end{notebox}

\begin{defbox}[title={Convergence}]
An infinite series of the form $X(e^{j\omega}) = \sum_{n=-\infty}^{\infty} x[n]\,e^{-j\omega n}$\,\HandoutEqRef{3.10} may or may not converge. Let
\begin{equation}
X_K(e^{j\omega}) = \sum_{n=-K}^{K} x[n]\,e^{-j\omega n}.\MitraTagRef{3.24}
\end{equation}
denote the partial sum of the weighted complex exponentials in Eq.~\HandoutEqRef{3.10}, and define
\begin{equation}
X(e^{j\omega}) = \lim_{K\to\infty} X_K(e^{j\omega})
\end{equation}
when this limit exists (in the sense made precise below).
\end{defbox}

\begin{defbox}[title={Def}]
We say that $X_K(e^{j\omega})$ converges \textbf{uniformly} to $X(e^{j\omega})$---stronger than pointwise convergence---if for every $\varepsilon>0$, there exists an integer $K$ such that for all integers $L\ge K$, the partial sum $X_L(e^{j\omega})$ of Eq.~\HandoutEqRef{3.24} satisfies
\begin{equation}
\big|X(e^{j\omega}) - X_L(e^{j\omega})\big| < \varepsilon \quad \text{for all } \omega.
\end{equation}
\end{defbox}

\begin{factbox}[title={Property}]
If $x[n]$ is an \textbf{absolutely summable} sequence---that is, if Eq.~\HandoutEqRef{3.25} holds with
\begin{equation}
\sum_{n=-\infty}^{\infty} |x[n]| < \infty.\MitraTagRef{3.25}
\end{equation}
then, by the triangle inequality and $|e^{-j\omega n}|=1$,
\begin{equation*}
\big|X(e^{j\omega})\big| = \left|\sum_{n=-\infty}^{\infty} x[n]\,e^{-j\omega n}\right|
\le \sum_{n=-\infty}^{\infty} |x[n]|\,\big|e^{-j\omega n}\big|
= \sum_{n=-\infty}^{\infty} |x[n]| < \infty
\end{equation*}
for all $\omega$, so the DTFT $X(e^{j\omega})$ of $x[n]$ exists for every $\omega$.

\medskip
\noindent Eq.~\HandoutEqRef{3.25} is thus a \textbf{sufficient condition} for existence of the DTFT $X(e^{j\omega})$:

\noindent Moreover, since $|x[n]e^{-j\omega n}|=|x[n]|$ is independent of $\omega$, the Weierstrass $M$-test shows that the series of Eq.~\HandoutEqRef{3.10} converges \textbf{uniformly} for all $\omega$.
\end{factbox}

\HandoutMarkLeft{7}

\begin{defbox}[title={Def}]
We say that $X_K(e^{j\omega})$ converges in the \textbf{mean-square} to $X(e^{j\omega})$ if
\begin{equation}
\lim_{K\to\infty} \int_{-\pi}^{\pi} \big|X(e^{j\omega}) - X_K(e^{j\omega})\big|^2\,d\omega = 0.
\end{equation}
\end{defbox}

\begin{factbox}[title={Property}]
The DTFT will converge in the mean-square if the sequence $x[n]$ is \textbf{square-summable}:
\begin{equation}
\sum_{n=-\infty}^{\infty} |x[n]|^2 < \infty.
\end{equation}
\end{factbox}

\begin{factbox}[title={Property (Nesting of the $\ell_p(\mathbb{Z})$ spaces)}]
\begin{itemize}
  \item Every absolutely summable sequence is also square summable.
  \item Every square-summable sequence is bounded.
\end{itemize}
\begin{equation}
\ell_1(\mathbb{Z}) \subset \ell_2(\mathbb{Z}) \subset \ell_\infty(\mathbb{Z}).
\end{equation}

\HandoutMarkSubsection{8}{An Example, and Generalized DTFTs}

\begin{examplebox}[title={Example}]
\label{ex:one-over-n}
The sequence
\begin{equation}
x[n] = \begin{cases} 1/n, & n \ge 1 \\ 0, & n \le 0 \end{cases}
\end{equation}
is \textbf{not} absolutely summable, but \textbf{is} square summable (famous result by Euler):
\begin{equation}
\sum_{n=1}^{\infty} \left(\frac{1}{n}\right)^2 = \frac{\pi^2}{6}.
\end{equation}
So $x[n]\in\ell_2(\mathbb{Z})\setminus\ell_1(\mathbb{Z})$: absolute summability is \textbf{not necessary} for a sequence to admit a useful DTFT analysis in the mean-square sense (cf.\ the diagram in \S\ref{sec:dtft-convergence}).
\end{examplebox}

We can sometimes define DTFTs in cases where the DTFT does not actually converge (but still exists in a generalized sense).

\begin{examplebox}[title={Example}]
$x[n] = e^{j\omega_0 n}$:
\begin{equation}
X(e^{j\omega}) = \sum_{n=-\infty}^{\infty} e^{j(\omega_0-\omega)n} \qquad \text{does not converge.}
\end{equation}
But $x[n]$ still has a useful (generalized) DTFT:
\begin{equation}
X(e^{j\omega}) = \sum_{k=-\infty}^{\infty} 2\pi\,\delta(\omega - \omega_0 + 2\pi k).
\end{equation}
\end{examplebox}

\HandoutMarkLeft{9}

\begin{equation}
X(e^{j\omega}) = 2\pi\,
\underbrace{\delta(\omega-\omega_0)}_{\HandoutUnderbraceLabel{\text{Dirac delta function}}},
\quad -\pi\le\omega\le\pi,\ -\pi\le\omega_0\le\pi
\end{equation}
\begin{align*}
x[n] &= \frac{1}{2\pi}\int_{-\pi}^{\pi} 2\pi\,\delta(\omega-\omega_0)\,e^{j\omega n}\,d\omega = e^{j\omega_0 n}.
\end{align*}
