\HandoutMarkSection{1}{The Discrete Fourier Transform (DFT)}

\begin{defbox}[title={Def}]
The DFT is a mapping of a length-$N$ sequence $x[n]$, $0\le n\le N-1$, into a length-$N$ Fourier representation $X[k]$, $k=0,\dots,N-1$, according to
\begin{equation*}
X[k] = \sum_{n=0}^{N-1} x[n]\,e^{-j\frac{2\pi kn}{N}}. \MitraTag{5.7}
\end{equation*}
\end{defbox}

\noindent It is a \textbf{sampled version of the DTFT} for a finite-length sequence:\footnotemark{}
\begin{equation*}
X[k] = X(e^{j\omega})\Big|_{\omega=\frac{2\pi k}{N}}.
\end{equation*}
\footnotetext{See the DTFT definition in Lecture~5 (September 8): $X(e^{j\omega})=\sum_{n=-\infty}^{\infty}x[n]\,e^{-j\omega n}$ (Mitra 3.10).}
$\Rightarrow$ discrete in both time and frequency.

\begin{defbox}[title={Inverse DFT}]
\begin{equation*}
x[n] = \frac{1}{N}\sum_{k=0}^{N-1} X[k]\,e^{+j\frac{2\pi kn}{N}}, \qquad n=0,\dots,N-1.
\end{equation*}
\end{defbox}

\HandoutMarkSubsection{2}{Twiddle-Factor Notation}

With $W_N = e^{-j\frac{2\pi}{N}}$,
\begin{equation*}
\text{Rewrite the DFT:}\quad
X[k] = \sum_{n=0}^{N-1} x[n]\,W_N^{kn}, \qquad 0\le k\le N-1. \MitraTag{11.60}
\end{equation*}
\begin{equation*}
\text{Rewrite the Inverse DFT:}\quad
x[n] = \frac{1}{N}\sum_{k=0}^{N-1} X[k]\,W_N^{-kn}, \qquad 0\le n\le N-1.
\end{equation*}

\paragraph{Verification of the inverse formula (check that the IDFT is indeed the inverse of the DFT).}
\begin{align*}
X[\ell] &= \sum_{n=0}^{N-1} x[n]\,W_N^{\ell n} \\
&= \HandoutHighlightMath{$\displaystyle\sum_{n=0}^{N-1}$}\left[\frac{1}{N}\HandoutHighlightYellow{$\displaystyle\sum_{k=0}^{N-1}$} X[k]\,W_N^{-kn}\right] W_N^{\ell n} \\
&\color{black!50}= \HandoutHighlightMath{$\color{black!50}\displaystyle\sum_{n=0}^{N-1}$}\HandoutHighlightYellow{$\color{black!50}\displaystyle\sum_{k=0}^{N-1}$} \frac{1}{N}\,X[k]\,W_N^{-kn}\,W_N^{\ell n}
\qquad \text{(move all factors into $\textstyle\sum_k$)} \\
&\color{black!50}= \HandoutHighlightYellow{$\color{black!50}\displaystyle\sum_{k=0}^{N-1}$}\HandoutHighlightMath{$\color{black!50}\displaystyle\sum_{n=0}^{N-1}$} \frac{1}{N}\,X[k]\,W_N^{-kn}\,W_N^{\ell n}
\qquad \text{(swap the sums)} \\
&= \HandoutHighlightYellow{$\displaystyle\sum_{k=0}^{N-1}$} X[k]\,\underbrace{\left[\textcolor{red}{\symbf{\frac{1}{N}}}\HandoutHighlightMath{$\displaystyle\sum_{n=0}^{N-1}$} W_N^{-(k-\ell)n}\right]}_{\HandoutUnderbraceLabel{\textcolor{red}{(\ast)}}}.
\end{align*}

Summing the geometric series (recall $W_N = e^{-j\frac{2\pi}{N}}$, so $W_N^{-(k-\ell)} = e^{j\frac{2\pi(k-\ell)}{N}}$),
\begin{align*}
\textcolor{red}{(\ast)} &= \frac{1}{N}\,\frac{1-W_N^{-(k-\ell)N}}{1-W_N^{-(k-\ell)}}
= \frac{1}{N}\,\frac{1-e^{j2\pi(k-\ell)}}{1-e^{j\frac{2\pi(k-\ell)}{N}}} \\
&= \frac{1}{N}\,\frac{e^{j\pi(k-\ell)}\,\textcolor{violet}{\left(e^{-j\pi(k-\ell)}-e^{j\pi(k-\ell)}\right)}}{e^{j\frac{\pi(k-\ell)}{N}}\,\textcolor{violet}{\left(e^{-j\frac{\pi(k-\ell)}{N}}-e^{j\frac{\pi(k-\ell)}{N}}\right)}} \\
&= \frac{1}{N}\,\frac{\textcolor{violet}{\sin\!\big(\pi[k-\ell]\big)}\,e^{j\pi(k-\ell)}}{\textcolor{violet}{\sin\!\big(\frac{\pi}{N}[k-\ell]\big)}\,e^{j\frac{\pi(k-\ell)}{N}}}\qquad\footnotemark{}
\end{align*}
\footnotetext{The \textcolor{violet}{$\sin/\sin$} factor is the \emph{Dirichlet kernel} $D_N(\omega)=\dfrac{\sin\!\left(\omega\frac{N}{2}\right)}{\sin\!\left(\frac{\omega}{2}\right)}$ (Lecture~5, September~8, \S~2), at $\symbf{\omega=\dfrac{2\pi(k-\ell)}{N}}$.}
\HandoutMarkBeforeAlign{3}%
\begin{align*}
&= \begin{cases} 1, & k \equiv \ell \pmod{N}, \\ 0, & \text{else} \end{cases}\qquad\footnotemark{} \\
\intertext{%
  \footnotesize In this verification, $k,\ell\in\{0,\ldots,N-1\}$, so $k\equiv\ell\pmod{N}\iff k=\ell$.%
}\\
&= \delta\big[\HandoutMod{k-\ell}{N}\big].
\end{align*}
\footnotetext{The geometric-series formula is only valid when $W_N^{-(k-\ell)}\ne 1$, i.e., $\symbf{k \not\equiv \ell \pmod{N}}$; then $\sin(\pi[k-\ell])=0$ while $\sin(\frac{\pi}{N}[k-\ell])\ne 0$, so $\textcolor{red}{(\ast)}=0$. When $k \equiv \ell \pmod{N}$, $W_N^{-(k-\ell)}=1$, so every term $W_N^{-(k-\ell)n}=1$ and $\textcolor{red}{(\ast)}=\frac{1}{N}\cdot N=1$ directly (also the $0/0$ limit of the Dirichlet-kernel expression).}

\begin{defbox}[title={Modulo Operation $\HandoutMod{n}{N}$}]
$k = \HandoutMod{n}{N}$ if $k\in\{0,\dots,N-1\}$ and there exists an integer $\ell$ such that $n = k + \ell N$.
\end{defbox}

\begin{examplebox}[title={Example $\HandoutMod{n}{3}$ {\normalfont\small\textit{(see Mitra~2.3.1)}}}]
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccccccc}
$n$ & $-2$ & $-1$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
\hline
$\HandoutMod{n}{3}$ & $1$ & $2$ & $0$ & $1$ & $2$ & $0$ & $1$ & $2$ & $0$
\end{tabular}
\end{center}
\end{examplebox}

\noindent Therefore
\begin{equation*}
X[\ell] = \sum_{k=0}^{N-1} X[k]\,\delta\big[\HandoutMod{k-\ell}{N}\big] = X[\ell] \quad \text{for } \ell\in\{0,\dots,N-1\}. \HandoutQEDhfillMath
\end{equation*}

\begin{examplebox}[title={Example}, handout mark=4]
$g[n] = \cos\!\left(\dfrac{2\pi rn}{N}\right)$, $\;0\le n\le N-1$, with $r$ an integer, $0\le r\le N-1$.
\noindent We can write $g[n]$ as
\begin{align*}
g[n] &= \tfrac{1}{2}\left(e^{j\frac{2\pi rn}{N}} + e^{-j\frac{2\pi rn}{N}}\right)
\qquad\text{(Euler)} \\
&= \tfrac{1}{2}\left(W_N^{-rn} + W_N^{rn}\right)
\qquad\text{(recall $W_N=e^{-j\frac{2\pi}{N}}$)}.
\end{align*}
\begin{align*}
G[k] &= \sum_{n=0}^{N-1} g[n]\,W_N^{kn}
= \sum_{n=0}^{N-1} \tfrac{1}{2}\left(W_N^{-rn} + W_N^{rn}\right) W_N^{kn} \\
&= \frac{1}{2}\left(\sum_{n=0}^{N-1} W_N^{-rn} W_N^{kn} + \sum_{n=0}^{N-1} W_N^{rn} W_N^{kn}\right) \\
&= \frac{1}{2}\left(\sum_{n=0}^{N-1} W_N^{-(r-k)n} + \sum_{n=0}^{N-1} W_N^{-(-r-k)n}\right) \\
&= \frac{1}{2}\Big(\textcolor{red}{\symbf{N}}\,\delta\big[\HandoutMod{r-k}{N}\big] + \textcolor{red}{\symbf{N}}\,\delta\big[\HandoutMod{-r-k}{N}\big]\Big).
\end{align*}
For $0\le k\le N-1$:
\begin{equation*}
G[k] = \begin{cases} \dfrac{N}{2}, & k = r,\ \ k = N-r, \\[4pt] 0, & \text{else}, \end{cases}
\qquad \text{(unless $r=0$ or $r=N/2$)}\footnotemark{}
\end{equation*}
\footnotetext{Reading the two $\delta$'s for $k\in\{0,\ldots,N-1\}$:
\begin{itemize}[nosep,topsep=2pt,leftmargin=1.4em]
  \item $k=r$: $\HandoutMod{r-k}{N}=0$ in the first term.
  \item $k=N-r$: $\HandoutMod{-r-k}{N}=0$ in the second term, i.e.\ $k\equiv -r\pmod{N}$ (the representative in $\{0,\ldots,N-1\}$ is $N-r$).
\end{itemize}}
using $\HandoutMod{-r-k}{N} = \HandoutMod{N-r-k}{N}$. When $r=0$ or $r=N/2$, the two impulses coincide and $G[r] = N$.
\end{examplebox}

\subsection{Periodicity}

\begin{factbox}[title={Property}]
\begin{align*}
X[\ell N + k] &= X[k], \qquad k=0,\dots,N-1,\ \ \ell \text{ an integer}, \\
x[mN + n] &= x[n], \qquad n=0,\dots,N-1,\ \ m \text{ an integer},
\end{align*}
when the DFT / IDFT formulas are evaluated outside $0,\dots,N-1$.
\end{factbox}
Because
\begin{align*}
W_N^N &= e^{-j\frac{2\pi}{N}N} = e^{-j2\pi} = 1 \\
&\Rightarrow\quad W_N^{\ell N+k} = (1)^{\ell}\,W_N^{k} = W_N^{k}.
\end{align*}
\noindent So adding $\ell N$ to a frequency index (or $mN$ to a time index) does not change any twiddle factor in the DFT/IDFT sums.
We therefore extend $X[k]$ and $x[n]$ periodically with period $N$ when those formulas are used for integer indices outside $\{0,\dots,N-1\}$ (as stated in the Property box).

\subsection{Circular Shift}

Consider $x[n]$, $0\le n\le N-1$. The ordinary shift $x_1[n] = x[n-n_0]$ is \textbf{no longer defined} over $0\le n\le N-1$.
\begin{equation*}
\Rightarrow\ \text{circular shifts:}\qquad x_1[n] = x\big[\HandoutMod{n-n_0}{N}\big]. \MitraTag{2.25}
\end{equation*}

\begin{factbox}[title={Property (Circular Time-Shifting)}]
Let $g[n] \longleftrightarrow G[k]$ and $\tilde g[n] = g\big[\HandoutMod{n-n_0}{N}\big]$. Then
\begin{equation*}
\tilde G[k] = W_N^{kn_0}\,G[k] = e^{-j\frac{2\pi kn_0}{N}}\,G[k].
\end{equation*}
\end{factbox}

\HandoutMarkParagraph{7}{Proof.}
\begin{align*}
\tilde G[k] &= \sum_{n=0}^{N-1} g\big[\HandoutMod{n-n_0}{N}\big]\,W_N^{kn}
= \sum_{n=0}^{n_0-1} g[n+N-n_0]\,W_N^{kn} + \sum_{n=n_0}^{N-1} g[n-n_0]\,W_N^{kn}.
\end{align*}
Let $m = n+N-n_0$ ($n = m-N+n_0$) in the first sum and $p = n-n_0$ ($n=p+n_0$) in the second:
\begin{align*}
\tilde G[k] &= \sum_{m=N-n_0}^{N-1} g[m]\,W_N^{k(m-N+n_0)} + \sum_{p=0}^{N-1-n_0} g[p]\,W_N^{k(p+n_0)} \\
&= \sum_{n=0}^{N-1} g[n]\,W_N^{k(n+n_0)} \qquad \left(W_N^{-kN}=1\right) \\
&= W_N^{kn_0}\left[\sum_{n=0}^{N-1} g[n]\,W_N^{kn}\right]
= W_N^{kn_0}\,G[k]
= e^{-j\frac{2\pi kn_0}{N}}\,G[k]. && \HandoutQEDalign
\end{align*}

\HandoutMarkSubsection{8}{Convolution}

\subsubsection{Linear convolution and the DTFT}
\begin{equation*}
x[n]\circledast h[n] \ \overset{\text{DTFT}}{\longleftrightarrow}\ X(e^{j\omega})\,H(e^{j\omega}).
\end{equation*}
If $x[n]$ has length $N_1$ and $h[n]$ has length $N_2$, then $y[n] = x[n]\circledast h[n]$ has length $N_1+N_2-1$: it is possibly nonzero for $0\le n\le N_1+N_2-2$ and is zero otherwise.

\subsubsection{Circular Convolution (two length-$N$ sequences)}
\begin{defbox}[title={Def}]
\begin{equation*}
y_c[n] = \sum_{m=0}^{N-1} x[m]\,h\big[\HandoutMod{n-m}{N}\big], \qquad 0\le n\le N-1, \MitraTag{5.51}
\end{equation*}
\begin{equation*}
y_c[n] = x[n] \HandoutCircConv h[n] = h[n] \HandoutCircConv x[n]. \MitraTag{5.52}
\end{equation*}
\end{defbox}

\begin{factbox}[title={Convolution Theorem}]
If $y_c[n] = x[n] \HandoutCircConv h[n]$, then $Y_c[k] = X[k]\,H[k]$.
\end{factbox}

\HandoutMarkParagraph{9}{Proof.}
\begin{align*}
Y_c[k] &= \HandoutHighlightMath{$\displaystyle\sum_{n=0}^{N-1}$}\left[\HandoutHighlightYellow{$\displaystyle\sum_{m=0}^{N-1}$} x\big[\HandoutMod{n-m}{N}\big]\,h[m]\right] W_N^{kn} \\
&\color{black!50}= \HandoutHighlightMath{$\color{black!50}\displaystyle\sum_{n=0}^{N-1}$}\HandoutHighlightYellow{$\color{black!50}\displaystyle\sum_{m=0}^{N-1}$} x\big[\HandoutMod{n-m}{N}\big]\,h[m]\, W_N^{kn}
\qquad \text{(expand inner sum)} \\
&\color{black!50}= \HandoutHighlightYellow{$\color{black!50}\displaystyle\sum_{m=0}^{N-1}$}\HandoutHighlightMath{$\color{black!50}\displaystyle\sum_{n=0}^{N-1}$} x\big[\HandoutMod{n-m}{N}\big]\,h[m]\, W_N^{kn}
\qquad \text{(swap the sums)} \\
&= \HandoutHighlightYellow{$\displaystyle\sum_{m=0}^{N-1}$} h[m]\left[\HandoutHighlightMath{$\displaystyle\sum_{n=0}^{N-1}$} x\big[\HandoutMod{n-m}{N}\big]\,W_N^{kn}\right] \\
&= \left[\sum_{m=0}^{N-1} h[m]\,W_N^{km}\right] X[k] \qquad \text{(circular time-shift)} \\
&= H[k]\,X[k]. && \HandoutQEDalign
\end{align*}

\subsection{Linear Convolution via Circular Convolution}

If $x[n]$, $0\le n\le N_1-1$, and $h[n]$, $0\le n\le N_2-1$, then
\begin{equation*}
y[n] = x[n]\circledast h[n], \qquad 0\le n\le N_1+N_2-2.
\end{equation*}

\begin{factbox}[title={Property}]
If
\begin{equation*}
\tilde x[n] = \begin{cases} x[n], & 0\le n\le N_1-1, \\ 0, & N_1\le n< N, \end{cases}
\qquad \text{where } N\ge N_1+N_2-1,
\end{equation*}
and similarly define a ``zero-padded'' length-$N$ $\tilde h[n]$, then
\begin{equation*}
\tilde x[n] \HandoutCircConv \tilde h[n] = x[n]\circledast h[n] \qquad \text{for } 0\le n\le N-1.
\end{equation*}
\end{factbox}

\begin{factbox}[title={Consequence: we can compute $\circledast$ using $\HandoutCircConv$!}, handout mark=10]
\begin{enumerate}
  \item Zero-pad both sequences to the same length $N\ge N_1+N_2-1$.
  \item Take length-$N$ DFTs of both.
  \item Multiply the DFTs together.
  \item Take the inverse DFT of the result.
\end{enumerate}
\end{factbox}

\begin{notebox}[title=\HandoutNoteAddendumTitle]
Why $N\ge N_1+N_2-1$: circular convolution equals the linear convolution \emph{aliased} with period $N$,
$\tilde x[n]\HandoutCircConv\tilde h[n] = \sum_{r} y[n+rN]$ for $0\le n\le N-1$. Since $y[n]$ occupies $N_1+N_2-1$ samples, no shifted copies overlap exactly when $N\ge N_1+N_2-1$.
With a fast algorithm (FFT) for the DFT, this is how long convolutions are computed in practice (cf.\ Mitra Ch.~11).
\end{notebox}

\subsection{Parseval's Theorem}

\begin{factbox}[title={\HandoutTitleWithFnmark{Property}}]
\begin{align*}
\sum_{n=0}^{N-1} x^*[n]\,y[n] &= \frac{1}{N}\sum_{k=0}^{N-1} X^*[k]\,Y[k], \\
&\Rightarrow\quad \sum_{n=0}^{N-1} |x[n]|^2 = \frac{1}{N}\sum_{k=0}^{N-1} |X[k]|^2. \MitraTag{5.6}
\end{align*}
\end{factbox}
\footnotetext{Notation vs.\ Mitra:
\begin{itemize}[nosep,topsep=2pt,leftmargin=1.4em]
  \item Sec.~5.1 is for a general orthogonal transform with basis $\psi[k,n]$. Its coefficients are written $\mathcal{X}[k]$ (calligraphic). $X[k]$ is used only for the DFT, from Sec.~5.2 on.
  \item Sec.~5.1 has only one sequence. Mitra (5.6) is the energy form $\sum_n|x[n]|^2=\frac{1}{N}\sum_k|\mathcal{X}[k]|^2$.
  \item Its derivation puts the conjugate on the second factor: $\sum_n x[n]\,x^*[n]=\frac{1}{N}\sum_k \mathcal{X}[k]\,\mathcal{X}^*[k]$.
  \item The two-sequence line above is Mitra (5.115) in Sec.~5.7: $\sum_n g[n]\,h^*[n]=\frac{1}{N}\sum_k G[k]\,H^*[k]$.
  \item This lecture writes $x,y$ instead of $g,h$ and puts the conjugate on the first factor. Both sides are then the complex conjugates of Mitra's, so the equation still holds.
\end{itemize}}

\subsection{Symmetry}

\begin{factbox}[title={Property}]
If $x[n]$ is real,
\begin{equation*}
X[k] = X^*[N-k] = X^*\big[\HandoutMod{-k}{N}\big], \qquad 0\le k\le N-1.
\end{equation*}
\end{factbox}
\noindent(For $k=0$, read $X^*[N]$ as $X^*[0]$ by periodicity; the modulo form handles this automatically.)
